   147→   147→function OverviewTab({ isMainAdmin, user }: { isMainAdmin: boolean; user: any }) {

   150→   150→  const [collegeFilter, setCollegeFilter] = useState("");
   151→   151→  const [lastUpdated, setLastUpdated] = useState<Date>(new Date());
   152→   152→
   153→   153→  const load = useCallback(async () => {
   154→   154→    try {
   155→   155→      const url = collegeFilter ? `/api/admin/stats?collegeId=${collegeFilter}` : "/api/admin/stats";
   156→   156→      const res = await fetch(url);
   157→   157→      const data = await res.json();
   158→   158→      if (res.ok) {
   159→   159→        setStats(data);
   160→   160→        setLastUpdated(new Date());

   163→   163→      // ignore
   164→   164→    } finally { setLoading(false); }
   165→   165→  }, [collegeFilter]);

   167→   167→  useEffect(() => { load(); }, [load]);
   168→   168→
   169→   169→  // Auto-refresh every 15 seconds
   170→   170→  useEffect(() => {
   171→   171→    const interval = setInterval(() => {
   172→   172→      load();

   200→   200→  const c = stats.counts;

   202→   202→  const topStudents = stats.topStudents || [];
   203→   203→  const recentActivity = stats.recentActivity || [];

   216→   216→          {isMainAdmin && (

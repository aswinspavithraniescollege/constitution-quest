   344→// ============== COLLEGES ==============
   345→function CollegesTab({ colleges, onRefresh }: any) {
   346→  const [name, setName] = useState("");
   347→  const [code, setCode] = useState("");
   348→  const [location, setLocation] = useState("");
   349→  const [loading, setLoading] = useState(false);
   350→  const [deletingId, setDeletingId] = useState<string | null>(null);
   351→  const [editing, setEditing] = useState<any | null>(null);
   352→
   353→  async function add(e: React.FormEvent) {
   354→    e.preventDefault();
   355→    if (!name || !code) return;
   356→    setLoading(true);
   357→    try {
   358→      const res = await fetch("/api/admin/colleges", {
   359→        method: "POST",
   360→        headers: { "Content-Type": "application/json" },
   361→        body: JSON.stringify({ name, code, location }),
   362→      });
   363→      const data = await res.json();
   364→      if (!res.ok) { toast.error(data.error); return; }
   365→      toast.success("College added!");
   366→      setName(""); setCode(""); setLocation("");
   367→      onRefresh();
   368→    } finally { setLoading(false); }
   369→  }
   370→
   371→  async function del(id: string, name: string) {
   372→    if (!confirm(`Delete "${name}"?\n\nThis will permanently remove the college. You can only delete a college that has no departments and no users (mentors, students, or sub-admins).`)) return;
   373→    setDeletingId(id);
   374→    try {
   375→      const res = await fetch(`/api/admin/colleges?id=${id}`, { method: "DELETE" });
   376→      const data = await res.json();
   377→      if (!res.ok) {
   378→        toast.error(data.error || "Failed to delete college");
   379→        return;
   380→      }
   381→      toast.success("College deleted");
   382→      onRefresh();
   383→    } finally {
   384→      setDeletingId(null);
   385→    }
   386→  }
   387→
   388→  async function saveEdit(e: React.FormEvent) {
   389→    e.preventDefault();
   390→    if (!editing) return;
   391→    setLoading(true);
   392→    try {
   393→      const res = await fetch("/api/admin/colleges", {
   394→        method: "PATCH",
   395→        headers: { "Content-Type": "application/json" },
   396→        body: JSON.stringify({ id: editing.id, name: editing.name, code: editing.code, location: editing.location }),
   397→      });
   398→      const data = await res.json();
   399→      if (!res.ok) { toast.error(data.error); return; }
   400→      toast.success("College updated!");
   401→      setEditing(null);
   402→      onRefresh();
   403→    } finally { setLoading(false); }
   404→  }
   405→
   406→  return (
   407→    <div className="grid grid-cols-1 lg:grid-cols-3 gap-6">
   408→      <div className="lg:col-span-1">

   519→      const res = await fetch(url);
